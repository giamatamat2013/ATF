enum Shape { Circle(float), Rect(float, float) }

fn area(s: Shape) -> float {
    return match s {
        Shape.Circle(r) => 3.14 * r * r
        Shape.Rect(w, h) => w * h
    }
}

fn main() {
    let shapes = [Shape.Circle(1.0), Shape.Rect(2.0, 3.0)]
    var total = 0.0
    for s in shapes {
        total += area(s)
    }
    println("total = {total:.2f}")
}
